```dart
import 'package:duxt_orm/duxt_orm.dart';

class User extends Model {
  @override
  String get tableName => 'users';

  @override
  Map<String, ColumnType> get schema => {
      'email': ColumnType.text,
      'password': ColumnType.text,
      'name': ColumnType.text,
  };

  String? email;
  String? password;
  String? name;

  @override
  void fromMap(Map<String, dynamic> map) {
    super.fromMap(map);
    email = map['email'] as String?;
    password = map['password'] as String?;
    name = map['name'] as String?;
  }

  @override
  Map<String, dynamic> toMap() {
    return {
      ...super.toMap(),
      'email': email,
      'password': password,
      'name': name,
    };
  }
}
```

### API route pattern:
```dart
import 'package:duxt/duxt.dart';

void registerApi(DuxtServer server) {
  server.get('/api/todo', (req) async {
    return Response.json({'message': 'Todo list'});
  });

  server.post('/api/todo', (req) async {
    final body = await req.body();
    return Response.json({'message': 'Todo created', 'data': body});
  });

  server.get('/api/todo/:id', (req) async {
    final id = req.params['id'];
    return Response.json({'message': 'Todo detail', 'id': id});
  });

  server.put('/api/todo/:id', (req) async {
    final id = req.params['id'];
    final body = await req.body();
    return Response.json({'message': 'Todo updated', 'id': id, 'data': body});
  });

  server.delete('/api/todo/:id', (req) async {
    final id = req.params['id'];
    return Response.json({'message': 'Todo deleted', 'id': id});
  });
}
```
