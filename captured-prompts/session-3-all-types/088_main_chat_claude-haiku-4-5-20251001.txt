   434→Middleware csrf({String headerName = 'X-CSRF-Token', String cookieName = 'csrf_token'}) {
   435→  return (Handler innerHandler) {
   436→    return (Request request) async {
   437→      final safeMethods = {'GET', 'HEAD', 'OPTIONS'};
   438→      if (safeMethods.contains(request.method)) {
   439→        return innerHandler(request);
   440→      }

   442→      final cookieHeader = request.headers['cookie'] ?? '';
   443→      final cookies = _parseCookies(cookieHeader);
   444→      final cookieToken = cookies[cookieName];
   445→      final headerToken = request.headers[headerName.toLowerCase()];
   446→
   447→      if (cookieToken == null || cookieToken.isEmpty ||
   448→          headerToken == null || headerToken != cookieToken) {
   449→        return Response.forbidden(
   450→          jsonEncode({'error': 'CSRF token mismatch'}),
   451→          headers: {'Content-Type': 'application/json'},
   452→        );
   453→      }

   455→      return innerHandler(request);

   457→  };
